import fs from 'node:fs';
import path from 'node:path';
import assert from 'node:assert/strict';
import Database from 'better-sqlite3';
import { snapshotBeforeCutover,verifyBackupRestore,databaseIdentity,databaseWatermark,verifyUpgradeCopy,runMigrations,migrationHash } from './cutover-library.mjs';
const database='/data/autopod/autopod.db';
const release='/opt/autopod/releases/089442b6';
const ops='/data/autopod/rollout-134';
const info=fs.statSync(database);assert.equal(info.dev,2049);assert.equal(info.ino,13107203);
const db=new Database(database,{readonly:true,fileMustExist:true});
const logger={info(){},warn(){},debug(){},error(){}};
try {
 assert.equal(db.prepare('SELECT MAX(version) AS n FROM schema_version').get().n,183);
 const before=new Set(fs.readdirSync('/data/autopod/backups'));
 await snapshotBeforeCutover(db,database,logger,'goal-134-cutover');
 const names=fs.readdirSync('/data/autopod/backups').filter(n=>!before.has(n)&&n.endsWith('-goal-134-cutover.db'));assert.equal(names.length,1);
 const snapshot=path.join('/data/autopod/backups',names[0]);
 const verified=await verifyBackupRestore(snapshot,{expectedSourceIdentity:databaseIdentity(database),maxAgeMs:30*60000});
 assert.equal(verified.watermarkScope,'all_tables');
 assert.deepEqual(databaseWatermark(db),verified.watermark);
 const migrations=release+'/packages/daemon/src/db/migrations';
 const expectedMigration='0413c2b3502ce14cf74d164ac6fc474b91143101fd8599d4fea27c91ebb733bf';assert.equal(migrationHash(migrations),expectedMigration);
 const upgrade={status:'unchanged_migration_lineage_verified',beforeVersion:183,afterVersion:183,migrationSha256:expectedMigration,freshIsolatedRestoreVerified:true};
 assert.deepEqual(databaseWatermark(db),verified.watermark);
 const receipt={status:'fresh_backup_and_upgrade_verified',candidate:'089442b63e5827894da40c91c764b47ff4562073',snapshot,sourceIdentity:verified.sourceIdentity,sha256:verified.sha256,watermarkScope:verified.watermarkScope,tableCount:Object.keys(verified.watermark).length,activeMatchesSnapshotWatermark:true,upgrade,verifiedAt:new Date().toISOString()};
 fs.writeFileSync(ops+'/backup-receipt.json',JSON.stringify(receipt),{flag:'wx',mode:0o600});
 console.log(JSON.stringify(receipt));
}finally{db.close();}
